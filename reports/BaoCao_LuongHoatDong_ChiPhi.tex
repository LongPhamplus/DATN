\section{Ứng dụng làm gì}

AAPEF mô phỏng việc \textbf{lập kế hoạch tấn công} của Red Team như một bài toán tìm đường trên không gian trạng thái, hoàn toàn bằng dữ liệu JSON (không máy ảo, không khai thác thật). Ứng dụng chạy \textbf{năm thuật toán} (BFS, DFS, Dijkstra, Greedy, A$^{*}$) trên cùng một kịch bản và so sánh: độ dài đường đi $L$, chi phí $C$, công sức tìm kiếm (số trạng thái mở rộng $E_n$, số cạnh xét $E_e$) và thời gian $T$.

\section{Giải thích ký hiệu}
\label{sec:symbols}

\textbf{Lưu ý chống nhầm:} chữ $E$ xuất hiện ở ba nghĩa khác nhau. $E$ (không có chỉ số) là \emph{điểm dễ khai thác}; $\E$ là \emph{số cạnh}; $E_n$ và $E_e$ là \emph{công sức tìm kiếm}.

\textbf{Kết quả của một lần chạy} (càng thấp càng tốt, trừ khi nói khác):

\begin{center}\small
\begin{tabular}{@{}Y{2.6cm}Y{13.2cm}@{}}
\toprule
Ký hiệu & Nghĩa đơn giản \\
\midrule
$L$ & Số bước (hành động) trong đường tấn công tìm được. \\
$C$ & Tổng chi phí của đường đó (cộng chi phí từng bước). Thấp nghĩa là đường dễ thành công hơn. \\
$\Ls$ & Số bước \emph{ít nhất} có thể có (lấy từ BFS). Dấu $*$ nghĩa là ``giá trị tốt nhất''. \\
$\Cs$ & Chi phí \emph{thấp nhất} có thể có (lấy từ Dijkstra); dùng làm mốc để so sánh. \\
$C/\Cs$ & Đường tìm được đắt gấp bao nhiêu lần mức tối ưu; $1{,}00$ là tối ưu. \\
$L-\Ls$ & Đường tìm được dài hơn mức tối thiểu bao nhiêu bước; $0$ là ngắn nhất. \\
$\V$ & Số trạng thái khác nhau có thể đạt tới; cho biết bài toán lớn cỡ nào. \\
$\E$ & Số cạnh, tức số bước đi có thể giữa các trạng thái. \\
$E_n$ & Số trạng thái thuật toán phải ``mở ra'' để xem xét. Càng ít thì càng ít công sức. \\
$E_e$ & Số cạnh thuật toán đã xét trong lúc mở các trạng thái. \\
$T$ & Thời gian chạy của phần tìm kiếm (mili giây). \\
tối ưu & Đạt đúng mức tốt nhất ($C=\Cs$ hoặc $L=\Ls$). \\
TB, trung vị, max & Trung bình; giá trị ở giữa khi xếp từ nhỏ đến lớn; giá trị lớn nhất. \\
$L\neq C$ & Đường ít bước nhất không phải là đường rẻ nhất. \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Ký hiệu trong công thức chi phí:}

\begin{center}\small
\begin{tabular}{@{}Y{2.6cm}Y{13.2cm}@{}}
\toprule
Ký hiệu & Nghĩa đơn giản \\
\midrule
$E$ & Điểm \emph{dễ khai thác} của CVSS: càng cao thì lỗ hổng càng dễ bị lợi dụng. \\
AV & \emph{Attack Vector}: tấn công từ xa qua mạng (N) hay phải ở ngay trên máy (L). \\
AC & \emph{Attack Complexity}: độ phức tạp, thấp (L) hoặc cao (H). \\
PR & \emph{Privileges Required}: quyền cần có trước khi tấn công: không (N), thấp (L), cao (H). \\
UI & \emph{User Interaction}: có cần người dùng thao tác không: không (N) hay cần (R). \\
S & \emph{Scope}: phạm vi ảnh hưởng, không đổi (U) hay đổi (C); chỉ làm thay đổi cách tính PR. \\
$p$ & Xác suất thành công của một bước, lấy bằng $E/10$. \\
cost & Chi phí của một bước, làm tròn $10\cdot(-\ln p)$. Bước dễ thì $p$ lớn nên cost nhỏ. \\
CVE & Mã định danh công khai của một lỗ hổng (ví dụ CVE-2021-44228). \\
NVD & Cơ sở dữ liệu lỗ hổng của NIST, nơi lấy vector CVSS của từng CVE. \\
KEV & Danh mục của CISA gồm các lỗ hổng đã bị khai thác ngoài thực tế. \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Năm thuật toán, nói ngắn:} BFS duyệt theo tầng nên ít bước nhất; DFS đi sâu trước nên không bảo đảm gì và phụ thuộc thứ tự khai báo hành động; Dijkstra luôn mở rộng trạng thái đang có tổng chi phí thấp nhất nên cho chi phí thấp nhất; Greedy chỉ nhìn ước lượng ``còn thiếu bao nhiêu để tới đích'' nên ít công sức nhưng có thể bị dẫn sai; A$^{*}$ là Dijkstra có thêm ước lượng đó, nên vẫn tối ưu mà mở rộng ít trạng thái hơn.

\section{Cách tính chi phí}
\label{sec:cost}

Chi phí \textbf{không do người thiết kế gõ}, mà được tính bằng công thức công khai của CVSS v3.1 (FIRST), áp dụng cho từng hành động chiếm quyền:
\[
E=8{,}22\times AV\times AC\times PR\times UI,\qquad p=\frac{E}{10},\qquad \text{cost}=\mathrm{round}\bigl(10\cdot(-\ln p)\bigr).
\]
$E$ là điểm \emph{Exploitability} (mức dễ khai thác); $p$ coi như xác suất thành công của bước. Phép $-\ln$ biến tích xác suất thành tổng chi phí, nên Dijkstra tìm \emph{đường có xác suất thành công cao nhất}. Chi phí càng thấp thì bước càng dễ.

\textbf{Vector CVSS (AV, AC, PR, UI, Scope) lấy từ hai nguồn:}
\begin{itemize}
  \item \textbf{Dẫn xuất từ cấu trúc hành động} (15 kịch bản): AV = Network nếu hành động cần đường mạng (\texttt{reachable}), ngược lại Local; PR theo mức quyền hành động yêu cầu (\texttt{none}, \texttt{user}, \texttt{admin}); AC = Low, UI = None (mặc định). Chỉ là xấp xỉ.
  \item \textbf{CVE thật} (kịch bản \texttt{lab-01}): hành động ghi trường \texttt{cve}; vector lấy từ bản ghi NVD trong \texttt{scenarios/cve\_catalog.json} (10 CVE thuộc danh mục CISA KEV). Điểm Exploitability tính từ vector trùng điểm NVD công bố ở 10/10 bản ghi.
\end{itemize}
Hành động chỉ thu thập thông tin (không nâng quyền) không phải lỗ hổng nên dùng $p=0{,}9$, chi phí $1$. Chi phí được gán một lần khi chạy \texttt{recost} (\texttt{aapef/cvss.py}), kết quả ghi vào \texttt{scenarios/cvss/}.

\begin{center}\small
\begin{tabular}{@{}llccc@{}}
\toprule
Ví dụ & Vector (AV/AC/PR/UI) & $E$ & $p$ & cost \\
\midrule
Từ xa, không cần quyền (Struts, Log4Shell) & N/L/N/N & 3,887 & 0,389 & 9 \\
Từ xa, quyền thấp (SMBv1) & N/L/L/N & 2,835 & 0,284 & 13 \\
Cục bộ, quyền thấp (PwnKit) & L/L/L/N & 1,835 & 0,184 & 17 \\
Cục bộ, phức tạp cao (Dirty COW) & L/H/L/N & 1,048 & 0,105 & 23 \\
Thu thập thông tin & --- & --- & 0,9 & 1 \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Giả định cần nêu khi bảo vệ:} Scope Unchanged (trừ khi CVE khai báo khác); chuẩn hóa $p=E/10$ và hệ số $10$ là lựa chọn thiết kế (có phân tích độ nhạy); các bước độc lập; CVSS chấm lỗ hổng chứ không chấm hành động, và đo mức dễ khai thác chứ không phải rủi ro. Chi phí vì vậy \emph{không} phản ánh độ khó thực của một cuộc tấn công cụ thể. Các kết luận chính bền vững trước 7 biến thể giả định (hệ số, chuẩn hóa, xác suất thu thập, AC và UI mặc định).

\section{Kết quả chính (tóm tắt)}

\begin{itemize}
  \item Dijkstra và A$^{*}$ luôn đạt chi phí tối ưu $\Cs$; BFS đạt số bước tối thiểu $\Ls$. Mọi đường đi đều hợp lệ khi phát lại.
  \item Với chi phí dẫn xuất từ cấu trúc, đường ít bước cũng là đường rẻ nhất: BFS, Dijkstra, A$^{*}$ cho cùng chi phí, \textbf{đánh đổi $L\neq C$ không xuất hiện}. Ở \texttt{lab-01} (chi phí từ CVE thật) nó xuất hiện: đường 5 bước tốn $43$, đường tối ưu 6 bước tốn $30$.
  \item DFS nhạy với thứ tự khai báo hành động (chi phí tới khoảng $8\times\Cs$); Greedy có thể bị bẫy ($30$ so với $14$ ở \texttt{trap-01/02}).
  \item Công sức của Dijkstra phụ thuộc số hành động rẻ độc lập hơn là $\V$; A$^{*}$ mở rộng ít hơn nhiều (tỉ lệ so với Dijkstra từ $0{,}29$ xuống $0{,}006$).
\end{itemize}

  \item \textbf{Độ nhạy thứ tự:} DFS đổi đường đi ở 11/12 kịch bản và có thể tốn tới $8\times\Cs$; BFS và Greedy chỉ nhạy khi có nhiều đường cùng độ dài hoặc cùng heuristic; Dijkstra và A$^{*}$ không bao giờ lệch khỏi $\Cs$.
  \item \textbf{Lưu ý khi đọc:} đây là thống kê \emph{mô tả} trên 12 kịch bản cùng một họ cấu trúc, trong đó \texttt{lab-01} do tác giả thiết kế; không dùng để suy diễn thống kê cho mạng thực. Tổng $E_n$ bị chi phối bởi một kịch bản lớn nên nên đọc cùng trung vị.
\end{itemize}
